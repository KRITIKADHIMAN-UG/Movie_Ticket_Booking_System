├── 01_Movie.cpp            # Base details (Title, language, duration)
├── 02_Seat.cpp             # Layout model defining physical number and base tier
├── 03_Screen.cpp           # Complex layout engine containing core physical seats
├── 04_Cinema.cpp           # High-level shell containing localized screen vectors
├── 05_Show.cpp             # Relational link handling Movie, Screen, and dynamic ShowSeats
├── 06_ShowSeat.cpp         # Contextual wrapper holding individual seat allocation flags
├── 07_Customer.cpp         # Customer contact detail model
├── 08_Booking.cpp          # Invoice record holding unique IDs, metrics, and state flags
├── 09_Payment.cpp          # Pure virtual base defining payment contracts
├── 10_PaymentTypes.cpp     # Explicit concrete integrations for UPI, Card, and Cash
├── 11_PriceCalculator.cpp  # Dedicated functional module calculating tiered rates
├── 12_TicketPrinter.cpp    # Direct terminal representation engine
├── 13_BookingService.cpp   # Global workflow orchestrator managing state loops
└── main.cpp                # Terminal interface management and input parsing
